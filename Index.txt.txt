// ============================================================
// State

let videoStream = null;
let uploadedImageBase64 = null;
let currentResults = [];

// ============================================================
// Tab switching
// ============================================================
function switchTab(tab) {
  document.querySelectorAll('.panel').forEach(p => p.classList.remove('active'));
  document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
  document.getElementById(tab + '-panel').classList.add('active');
  document.getElementById('tab-' + tab).classList.add('active');
  clearResults();
}

// ============================================================
// Camera
// ============================================================
async function startCamera() {
  try {
    const constraints = {
      video: {
        facingMode: { ideal: 'environment' },
        width: { ideal: 1280 },
        height: { ideal: 960 }
      }
    };
    videoStream = await navigator.mediaDevices.getUserMedia(constraints);
    const video = document.getElementById('video');
    video.srcObject = videoStream;
    video.style.display = 'block';
    document.getElementById('cam-placeholder').style.display = 'none';
    document.getElementById('canvas-overlay').style.display = 'block';
    document.getElementById('cam-start-btn').style.display = 'none';
    document.getElementById('scan-btn').disabled = false;
    setStatus('카메라 활성 — 프로그램을 비추고 스캔하세요', 'active');
  } catch (e) {
    setStatus('카메라 접근 불가: ' + e.message, '');
  }
}

function setStatus(msg, cls) {
  const s = document.getElementById('cam-status');
  s.textContent = msg;
  s.className = 'cam-status' + (cls ? ' ' + cls : '');
}

async function captureAndScan() {
  const video = document.getElementById('video');
  const canvas = document.createElement('canvas');
  canvas.width = video.videoWidth || 640;
  canvas.height = video.videoHeight || 480;
  const ctx = canvas.getContext('2d');
  ctx.drawImage(video, 0, 0);
  const base64 = canvas.toDataURL('image/jpeg', 0.85).split(',')[1];
  
  document.getElementById('scan-frame').classList.add('show');
  setStatus('스캔 중...', 'scanning');
  document.getElementById('scan-btn').disabled = true;
  
  await analyzeImage(base64, 'camera');
  
  document.getElementById('scan-frame').classList.remove('show');
  setStatus('카메라 활성', 'active');
  document.getElementById('scan-btn').disabled = false;
}

// ============================================================
// File upload
// ============================================================
function handleFileSelect(event) {
  const file = event.target.files[0];
  if (!file) return;
  
  const reader = new FileReader();
  reader.onload = (e) => {
    const result = e.target.result;
    uploadedImageBase64 = result.split(',')[1];
    document.getElementById('preview-img').src = result;
    document.getElementById('preview-wrap').style.display = 'block';
    clearResults();
  };
  reader.readAsDataURL(file);
}

// Drag and drop
const dz = document.getElementById('drop-zone');
dz.addEventListener('dragover', e => { e.preventDefault(); dz.classList.add('dragover'); });
dz.addEventListener('dragleave', () => dz.classList.remove('dragover'));
dz.addEventListener('drop', e => {
  e.preventDefault();
  dz.classList.remove('dragover');
  const file = e.dataTransfer.files[0];
  if (file && file.type.startsWith('image/')) {
    const reader = new FileReader();
    reader.onload = ev => {
      uploadedImageBase64 = ev.target.result.split(',')[1];
      document.getElementById('preview-img').src = ev.target.result;
      document.getElementById('preview-wrap').style.display = 'block';
      clearResults();
    };
    reader.readAsDataURL(file);
  }
});

async function analyzeUploadedImage() {
  if (!uploadedImageBase64) return;
  document.getElementById('analyze-btn').disabled = true;
  await analyzeImage(uploadedImageBase64, 'upload');
  document.getElementById('analyze-btn').disabled = false;
}

// ============================================================
// Manual search
// ============================================================
async function searchManual() {
  const q = document.getElementById('manual-input').value.trim();
  if (!q) return;
  clearResults();
  showLoading(true);
  try {
    const result = await queryClaudeManual(q);
    renderResults(result);
  } catch (e) {
    renderError(e.message);
  }
  showLoading(false);
}

// ============================================================
// Core: analyze image via Claude API
// ============================================================
async function analyzeImage(base64, source) {
  clearResults();
  showLoading(true);
  
  const prompt = `당신은 클래식 음악 전문가입니다. 이 이미지는 클래식 콘서트 프로그램북 또는 연주회 안내 자료입니다.

이미지에서 모든 곡 제목을 찾아내고, 각 곡의 일반적인 연주 시간을 알려주세요.

**중요 규칙:**

async function queryClaudeManual(query) {
  const prompt = `당신은 클래식 음악 전문가입니다.

사용자가 입력한 곡 또는 연주회 프로그램:
"${query}"

각 곡의 연주 시간을 알려주세요.

**중요 규칙:**
1. 악장 일부인 경우 해당 악장 시간만 계산하세요.
2. 여러 녹음이 존재하면 평균 시간을 사용하세요.
3. 모르는 곡은 솔직하게 "알 수 없음"으로 표시하세요.
4. 여러 곡이 나열되어 있으면 모두 개별로 처리하세요.

반드시 아래 JSON 형식으로만 응답하세요:
{
  "pieces": [
    {
      "title": "곡 제목",
      "composer": "작곡가",
      "duration_minutes": 숫자 또는 null,
      "duration_display": "약 XX분 또는 알 수 없음",
      "is_partial": true/false,
      "partial_detail": "",
      "note": "",
      "unknown": true/false,
      "movements": []
    }
  ]
}`;

  const response = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      model: "claude-sonnet-4-20250514",
      max_tokens: 1000,
      messages: [{ role: "user", content: prompt }]
    })
  });

  const data = await response.json();
  if (data.error) throw new Error(data.error.message);
  
  const text = data.content.map(c => c.text || '').join('');
  const clean = text.replace(/```json|```/g, '').trim();
  const parsed = JSON.parse(clean);
  return parsed.pieces;
}

// Render results
// ============================================================
function renderResults(pieces) {
  currentResults = pieces || [];
  const list = document.getElementById('results-list');
  
  if (!pieces || pieces.length === 0) {
    list.innerHTML = `<div class="empty-state"><div class="es-icon">🎵</div><p>곡을 찾지 못했습니다.<br>다시 시도해주세요.</p></div>`;
    return;
  }

  let header = `<div class="results-header">발견된 곡 · ${pieces.length}곡</div>`;
  let html = header;

  let totalMinutes = 0;
  let hasAllDurations = pieces.length > 0;

  pieces.forEach(p => {
    const unknownClass = p.unknown ? ' unknown' : '';
    let durationHtml;
    
    if (p.unknown || !p.duration_minutes) {
      durationHtml = `<div class="rc-duration unknown">연주 시간 알 수 없음</div>`;
      hasAllDurations = false;
    } else {
      const mins = Math.round(p.duration_minutes);
      totalMinutes += mins;
      const h = Math.floor(mins / 60);
      const m = mins % 60;
      const timeStr = h > 0 ? `${h}:${String(m).padStart(2,'0')}` : `${m}분`;
      durationHtml = `<div class="rc-duration">${timeStr}</div><div class="rc-meta">${p.duration_display}</div>`;
    }

    const partialBadge = p.is_partial 
      ? `<span class="rc-badge partial">${p.partial_detail || '일부'}</span>` 
      : '';

    let movementsHtml = '';
    if (p.movements && p.movements.length > 0) {
      const items = p.movements.map(m => 
        `<div class="rc-movement-item"><span>${m.name}</span><span>${m.duration_display}</span></div>`
      ).join('');
      movementsHtml = `<div class="rc-movements"><div class="rc-movements-title">악장 구성</div>${items}</div>`;
    }

    html += `
      <div class="result-card${unknownClass}">
        <div class="rc-piece">${p.title}${partialBadge}</div>
        <div class="rc-composer">${p.composer || ''}</div>
        <div class="rc-duration-row">
          ${durationHtml}
        </div>
        ${p.note ? `<div class="rc-note">※ ${p.note}</div>` : ''}
        ${movementsHtml}
      </div>`;
  });

  list.innerHTML = html;

  // Total bar
  if (pieces.length > 1 && totalMinutes > 0) {
    const th = Math.floor(totalMinutes / 60);
    const tm = totalMinutes % 60;
    const totalStr = th > 0 ? `${th}시간 ${tm}분` : `${tm}분`;
    document.getElementById('total-time').textContent = totalStr;
    document.getElementById('total-bar').classList.add('show');
    // append total bar before results
    document.getElementById('results-area').insertBefore(
      document.getElementById('total-bar'),
      list
    );
  }
}

// AR overlays (camera mode)
// ============================================================
function renderAROverlays(pieces) {
  const container = document.getElementById('ar-overlays');
  container.innerHTML = '';
  if (!pieces || pieces.length === 0) return;

  const vcHeight = document.getElementById('video-container').offsetHeight;
  const vcWidth = document.getElementById('video-container').offsetWidth;
  const spacing = vcHeight / (pieces.length + 1);

  pieces.forEach((p, i) => {
    if (!p || p.unknown) return;
    const box = document.createElement('div');
    box.className = 'ar-box';
    const mins = p.duration_minutes ? Math.round(p.duration_minutes) : null;
    const durText = mins ? (Math.floor(mins/60) > 0 ? `${Math.floor(mins/60)}:${String(mins%60).padStart(2,'0')}` : `${mins}분`) : '?';
    box.innerHTML = `
      <div class="ar-title">${p.title}</div>
      <div class="ar-duration">${durText}</div>
      <div class="ar-detail">${p.composer || ''}</div>
    `;
    // Position scattered across frame
    const top = spacing * (i + 1) - 30;
    const left = i % 2 === 0 ? 10 : vcWidth - 230;
    box.style.top = top + 'px';
    box.style.left = Math.max(5, Math.min(left, vcWidth - 230)) + 'px';
    container.appendChild(box);
  });
}
